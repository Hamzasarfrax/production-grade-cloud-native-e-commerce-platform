services:

  # Laravel application running on PHP-FPM
  backend-app:
    build:
      context: .
      dockerfile: Dockerfile

    image: backend-app:1.0.0

    container_name: backend-app

    restart: unless-stopped

    # PHP-FPM internal port
    expose:
      - "9000"

    # Laravel .env
    env_file:
      - .env

    # .env local dev ke liye hai; container ke andar DB host override
    environment:
      DB_HOST: mysql

    depends_on:
      mysql:
        condition: service_healthy

    networks:
      - backend-app-network


  # Nginx web server
  nginx:
    image: nginx:alpine

    container_name: backend-app-nginx

    restart: unless-stopped

    ports:
      - "8080:80"

    volumes:
      # Laravel application
      - ./:/backend-app:ro

      # Nginx configuration
      - ./docker/nginx/default.conf:/etc/nginx/conf.d/default.conf:ro

    depends_on:
      - backend-app

    networks:
      - backend-app-network


  # MySQL database
  mysql:
    image: mysql:8.4

    container_name: app_mysql

    restart: unless-stopped

    environment:
      MYSQL_DATABASE: ${DB_DATABASE}
      MYSQL_USER: ${DB_USERNAME}
      MYSQL_PASSWORD: ${DB_PASSWORD}
      MYSQL_ROOT_PASSWORD: ${DB_ROOT_PASSWORD}

    ports:
      - "3306:3306"

    volumes:
      - mysql-data:/var/lib/mysql

    healthcheck:
      test:
        [
          "CMD-SHELL",
          "mysqladmin ping -h localhost -u root -p\"$$MYSQL_ROOT_PASSWORD\" --silent"
        ]
      interval: 10s
      timeout: 5s
      retries: 5
      start_period: 30s

    networks:
      - backend-app-network


volumes:

  mysql-data:


networks:

  backend-app-network:
    driver: bridge
