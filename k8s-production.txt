 PROJECT ARCHITECTURE:
 --------------------
   Frontend (React 19 + Vite + Nginx)  :3000  ->  K8s Service  ->  Ingress

================================================================================
 PHASE 0: PREREQUISITES — Tool Installation (Local Machine)
================================================================================

 Step 0.1  Install kubectl (Kubernetes CLI)
 -------------------------------------------
   # Linux
   curl -LO "https://dl.k8s.io/release/$(curl -sL https://dl.k8s.io/release/stable.txt)/bin/linux/amd64/kubectl"
   chmod +x kubectl && sudo mv kubectl /usr/local/bin/

   # Windows (PowerShell)
   curl.exe -LO "https://dl.k8s.io/release/v1.31.0/bin/windows/amd64/kubectl.exe"

   # Verify
   kubectl version --client

 Step 0.2  Install Helm (Package Manager)
 ------------------------------------------
   curl https://raw.githubusercontent.com/helm/helm/main/scripts/get-helm-3 | bash
   helm version

 Step 0.3  Install Docker (Build Images)
 ------------------------------------------
   # Already installed (we have docker-compose.yml working)
   docker --version

 Step 0.4  Install a Local Cluster — Pick ONE:
 ------------------------------------------------
   Option A: kind (Kubernetes IN Docker) — LIGHTWEIGHT, RECOMMENDED FOR DEV
   ──────────
     curl -Lo ./kind https://kind.sigs.k8s.io/dl/latest/kind-linux-amd64
     chmod +x ./kind && sudo mv ./kind /usr/local/bin/kind

     kind create cluster --config k8s/kind.yaml --name mxmobilz
     # k8s/kind.yaml already has 1 control-plane + 2 workers

   Option B: minikube — SINGLE NODE, GOOD FOR LEARNING
   ──────────
     curl -LO https://storage.googleapis.com/minikube/releases/latest/minikube-linux-amd64
     sudo install minikube /usr/local/bin/minikube
     minikube start --driver=docker --cpus=4 --memory=8192

   Option C: k3s — LIGHTWEIGHT PROD-LIKE (Rancher)
   ──────────
     curl -sfL https://get.k3s.io | INSTALL_K3S_EXEC="--docker" sh -
     export KUBECONFIG=/etc/rancher/k3s/k3s.yaml

   Option D: Cloud Managed Cluster (PRODUCTION)
   ──────────
     AWS EKS:   eksctl create cluster --name mxmobilz --nodes 3 --node-type t3.medium
     GKE:       gcloud container clusters create mxmobilz --num-nodes=3 --machine-type=e2-medium
     AKS:       az aks create -g mxmobilz-rg -n mxmobilz --node-count 3 --node-vm-size Standard_B2s

 Step 0.5  Verify Cluster
 --------------------------
   kubectl cluster-info
   kubectl get nodes
   # 3 nodes should be Ready (1 control-plane + 2 workers for kind)

 PHASE 1: CONTAINER IMAGE BUILD & PUSH TO REGISTRY

 Step 1.1  Login to Docker Hub (or your private registry)
 -----------------------------------------------------------
   docker login
   # Enter username: hamzasarfraz862
   # Enter password/token

 Step 1.2  Build Backend Image (Multi-Stage)
 ----------------------------------------------
   docker build -t hamzasarfraz862/backend-app:1.0.2 ./backend
   # This uses the multi-stage Dockerfile:
   #   Stage 1 (vendor): composer:2.8 installs dependencies
   #   Stage 2 (prod):  php:8.3-fpm with pdo_mysql, opcache, zip, gd

 Step 1.3  Build Frontend Image (Multi-Stage)
 -----------------------------------------------
   docker build -t hamzasarfraz862/front-end-app:1.0.0 ./frontend
   # This uses the multi-stage Dockerfile:
   #   Stage 1 (builder): node:20-alpine builds Vite app
   #   Stage 2 (prod):   nginx:alpine serves static + /api reverse-proxy

 Step 1.4  Tag with Git SHA (Production Best Practice — traceable deploys)
 ---------------------------------------------------------------------------
   GIT_SHA=$(git rev-parse --short HEAD)
   docker tag hamzasarfraz862/backend-app:1.0.2  hamzasarfraz862/backend-app:$GIT_SHA
   docker tag hamzasarfraz862/front-end-app:1.0.0 hamzasarfraz862/front-end-app:$GIT_SHA

 Step 1.5  Push Images to Registry
 -------------------------------------
   docker push hamzasarfraz862/backend-app:1.0.2
   docker push hamzasarfraz862/backend-app:$GIT_SHA
   docker push hamzasarfraz862/front-end-app:1.0.0
   docker push hamzasarfraz862/front-end-app:$GIT_SHA

 Step 1.6  (Optional) Private Registry on Cluster
 ---------------------------------------------------
   # If using self-hosted registry instead of Docker Hub:
   # Deploy a registry inside K8s:
   kubectl create deployment registry --image=registry:2 -n kube-system
   kubectl expose deployment registry --port=5000 --type=ClusterIP -n kube-system
   # Then push images to: registry.kube-system.svc.cluster.local:5000/backend-app:tag

 PHASE 6: INGRESS (External Access — TLS + Routing)
================================================================================

 Step 6.1  Install NGINX Ingress Controller
 ----------------------------------------------
   # kind (already installed):
   kubectl apply -f https://raw.githubusercontent.com/kubernetes/ingress-nginx/main/deploy/static/provider/kind/deploy.yaml

   # Helm (production):
   helm repo add ingress-nginx https://kubernetes.github.io/ingress-nginx
   helm repo update
   helm install ingress-nginx ingress-nginx/ingress-nginx \
     --namespace ingress-nginx \
     --create-namespace \
     --set controller.replicaCount=2 \
     --set controller.resources.requests.memory=128Mi \
     --set controller.resources.requests.cpu=100m

   # Wait for controller:
   kubectl wait --namespace ingress-nginx \
     --for=condition=ready pod \
     --selector=app.kubernetes.io/component=controller \
     --timeout=120s

 Step 6.2  Create Ingress Resource
 -------------------------------------
   kubectl apply -f k8s/ingress/ingress.yaml
   # k8s/ingress/ingress.yaml content:

   #       cert-manager.io/cluster-issuer: "letsencrypt-prod"

 Step 6.3  Setup TLS with cert-manager (Let's Encrypt — FREE SSL)
 -------------------------------------------------------------------
   # Install cert-manager:
   helm repo add jetstack https://charts.jetstack.io
   helm repo update
   helm install cert-manager jetstack/cert-manager \
     --namespace cert-manager \
     --create-namespace \
     --set installCRDs=true

   # Create ClusterIssuer for Let's Encrypt:
   kubectl apply -f - <<EOF
   apiVersion: cert-manager.io/v1
   kind: ClusterIssuer
   metadata:
     name: letsencrypt-prod
   spec:
     acme:
       server: https://acme-v02.api.letsencrypt.org/directory
       email: your-email@domain.com
       privateKeySecretRef:
         name: letsencrypt-prod-key
       solvers:
         - http01:
             ingress:
               class: nginx
   EOF

   # Verify TLS certificate is issued:
   kubectl get certificate -n cloud-native-ecomerce-app
   kubectl describe certificate mxmobilz-tls -n cloud-native-ecomerce-app

 PHASE 7: NETWORK POLICIES (Security — Restrict Pod-to-Pod Traffic)
================================================================================

 Step 7.1  Create Network Policies
 ------------------------------------
   kubectl apply -f - <<EOF
   apiVersion: networking.k8s.io/v1
   kind: NetworkPolicy
   metadata:
     name: deny-all-ingress
     namespace: cloud-native-ecomerce-app
   spec:
     podSelector: {}
     policyTypes:
       - Ingress
   ---
   apiVersion: networking.k8s.io/v1
   kind: NetworkPolicy
   metadata:
     name: allow-frontend-to-backend
     namespace: cloud-native-ecomerce-app
   spec:
     podSelector:
       matchLabels:
         app: backend-app

 PHASE 13: BACKUP & DISASTER RECOVERY
================================================================================

 Step 13.1  MySQL Backup CronJob
 -----------------------------------
   kubectl apply -f - <<EOF
   apiVersion: batch/v1
   kind: CronJob
   metadata:
     name: mysql-backup
     namespace: cloud-native-ecomerce-app
   spec:
     schedule: "0 2 * * *"  # Daily at 2 AM
     jobTemplate:
       spec:
         template:
           spec:
             containers:
               - name: mysql-backup
                 image: mysql:8.4
                 command:
                   - /bin/sh
                   - -c
                   - |
                     mysqldump -h mysql -u root -p\$MYSQL_ROOT_PASSWORD \
                       cloud-native > /backup/dump-\$(date +%Y%m%d).sql
                 env:
                   - name: MYSQL_ROOT_PASSWORD
                     valueFrom:
                       secretKeyRef:
                         name: mysql-secret
                         key: mysql-root-password
                 volumeMounts:
                   - name: backup-volume
                     mountPath: /backup
             restartPolicy: OnFailure
             volumes:
               - name: backup-volume
                 persistentVolumeClaim:
                   claimName: backup-pvc
   EOF

 Step 13.2  Create Backup PVC
 ---------------------------------
   kubectl apply -f - <<EOF
   apiVersion: v1
   kind: PersistentVolumeClaim
   metadata:
     name: backup-pvc
     namespace: cloud-native-ecomerce-app
   spec:
     accessModes: [ReadWriteOnce]
     resources:
       requests:
         storage: 20Gi
   EOF

 Step 13.3  Velero (Cluster-Level Backup — Production Gold Standard)
 ----------------------------------------------------------------------
   # Install Velero CLI:
   curl -fsSL https://github.com/vmware-tanzu/velero/releases/latest/download/velero-v1.14.0-linux-amd64.tar.gz | tar xz
   sudo mv velero-v1.14.0-linux-amd64/velero /usr/local/bin/

   # Install Velero server (AWS S3 example):
   velero install \
     --provider aws \
     --plugins velero/velero-plugin-for-aws:v1.9.2 \
     --bucket mxmobilz-backups \
     --secret-file ./credentials-velero \
     --backup-location-config region=us-east-1 \
     --snapshot-location-config region=us-east-1

   # Create backup:
   velero backup create mxmobilz-full --include-namespaces cloud-native-ecomerce-app

   # Restore:
   velero restore create --from-backup mxmobilz-full
